# Builds the Scotch 7.0.17 library (libscotch, without PT-Scotch), with
# multithreading and deterministic behavior (see CHANGES_Neper).

cmake_minimum_required(VERSION 3.0...3.10)
project(scotch C)

find_package(Threads REQUIRED)

# Compilation flags (same as Scotch's own build system)
set(SCOTCH_DEFINITIONS COMMON_PTHREAD SCOTCH_PTHREAD SCOTCH_RENAME
    restrict=__restrict SCOTCH_VERSION_NUM=7 SCOTCH_RELEASE_NUM=0
    SCOTCH_PATCHLEVEL_NUM=17)
if(APPLE)
  list(APPEND SCOTCH_DEFINITIONS COMMON_OS_MACOS _DARWIN_C_SOURCE)
endif()
include(CheckSymbolExists)
set(CMAKE_REQUIRED_DEFINITIONS -D_GNU_SOURCE)
set(CMAKE_REQUIRED_LIBRARIES Threads::Threads)
check_symbol_exists(pthread_setaffinity_np pthread.h HAVE_PTHREAD_SETAFFINITY_NP)
unset(CMAKE_REQUIRED_DEFINITIONS)
unset(CMAKE_REQUIRED_LIBRARIES)
if(HAVE_PTHREAD_SETAFFINITY_NP)
  list(APPEND SCOTCH_DEFINITIONS COMMON_PTHREAD_AFFINITY_LINUX)
endif()
if(CMAKE_C_COMPILER_ID STREQUAL "GNU")
  set(CMAKE_C_FLAGS "-U_FORTIFY_SOURCE ${CMAKE_C_FLAGS}")  # known to cause problems
endif()

# scotch.h is generated by dummysizes from library.h, as it contains the
# sizes of the library structures on the target platform
add_executable(scotch_dummysizes src/libscotch/dummysizes.c)
target_compile_definitions(scotch_dummysizes PRIVATE ${SCOTCH_DEFINITIONS})
target_include_directories(scotch_dummysizes PRIVATE src/libscotch)
target_link_libraries(scotch_dummysizes PRIVATE Threads::Threads)
add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/scotch.h
  COMMAND scotch_dummysizes "-s" ${CMAKE_CURRENT_SOURCE_DIR}/src/libscotch/library.h
          ${CMAKE_CURRENT_BINARY_DIR}/scotch.h
  DEPENDS scotch_dummysizes src/libscotch/library.h
  COMMENT "Generating scotch.h")
add_custom_target(scotch_h DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/scotch.h)

add_dependencies(scotcherr scotch_h)

add_library(scotcherrexit src/libscotch/library_error_exit.c)
target_compile_definitions(scotcherrexit PRIVATE ${SCOTCH_DEFINITIONS})
add_dependencies(scotcherrexit scotch_h)

  target_compile_options(${target} PRIVATE -w)
  target_include_directories(${target} PRIVATE src/libscotch

endforeach()
